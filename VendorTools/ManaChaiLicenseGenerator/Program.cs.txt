internal sealed class KeyInfo
{
    public string KeyId { get; set; } = string.Empty;
    public string Algorithm { get; set; } = string.Empty;
    public string PublicKeySha256 { get; set; } = string.Empty;
    public DateTime CreatedAtUtc { get; set; }
}

internal sealed class LicensePayload
{
    public string SchemaVersion { get; set; } = "MCL-LIC-1";
    public string KeyId { get; set; } = string.Empty;
    public string LicenseId { get; set; } = string.Empty;
    public string CustomerName { get; set; } = string.Empty;
    public string MachineId { get; set; } = string.Empty;
    public string LicenseType { get; set; } = string.Empty;
    public DateTime IssuedAtUtc { get; set; }
    public DateTime? ExpiresAtUtc { get; set; }
}

internal sealed class SignedLicenseFile
{
    public LicensePayload Payload { get; set; } = new();
    public string SignatureAlgorithm { get; set; } = "RSA-PSS-SHA256";
    public string SignatureBase64 { get; set; } = string.Empty;
}

    private void RefreshKeyState()
    {
        bool exists =
            File.Exists(PrivateKeyPath) &&
            File.Exists(PublicKeyPath) &&
            File.Exists(KeyInfoPath);

        _generateButton.Enabled = exists;

        if (!exists)
        {
            _keyStatusValue.Text =
                "ไม่พบ Vendor Signing Key";
            _keyStatusValue.ForeColor =
                System.Drawing.Color.DarkRed;
            _keyIdValue.Text = "-";
            return;
        }

        try
        {
            KeyInfo info =
                LoadAndValidateKeyMetadata();

            _keyStatusValue.Text =
                "พร้อมออก License";
            _keyStatusValue.ForeColor =
                System.Drawing.Color.DarkGreen;
            _keyIdValue.Text =
                info.KeyId;
        }
        catch (Exception ex)
        {
            _keyStatusValue.Text =
                "Key ไม่พร้อมใช้งาน";
            _keyStatusValue.ForeColor =
                System.Drawing.Color.DarkRed;
            _keyIdValue.Text =
                ex.Message;
            _generateButton.Enabled = false;
        }
    }

    private void RefreshPreview()
    {
        string customer =
            string.IsNullOrWhiteSpace(

                : _customerNameBox.Text.Trim();

        string machine =
            string.IsNullOrWhiteSpace(
                _machineIdBox.Text)
                ? "(ยังไม่ระบุ Machine ID)"
                : NormalizeMachineId(
                    _machineIdBox.Text);

        string type =
            _trialRadio.Checked
                ? "Trial 7 วัน"
                : "Permanent";

        _licensePreviewValue.Text =
            $"ลูกค้า: {customer}\r\n" +
            $"Machine: {machine}   |   ประเภท: {type}";
    }

    private void GenerateLicense_Click(
        object? sender,
        EventArgs e)
    {
        string customerName =
            _customerNameBox.Text.Trim();

        string machineId =
            NormalizeMachineId(
                _machineIdBox.Text);

        if (string.IsNullOrWhiteSpace(
                customerName))
        {
            MessageBox.Show(
                "กรุณาระบุชื่อลูกค้า / ร้าน",
                "ManaChai License Generator",
                MessageBoxButtons.OK,
                MessageBoxIcon.Warning);

            _customerNameBox.Focus();
            return;
        }

        if (!MachineIdRegex.IsMatch(
                machineId))
        {
            MessageBox.Show(
                "Machine ID ไม่ถูกต้อง\n\n" +
                "รูปแบบต้องเป็น MC-XXXX-XXXX-XXXX",
                "ManaChai License Generator",
                MessageBoxButtons.OK,
                MessageBoxIcon.Warning);

            _machineIdBox.Focus();
            return;
        }

        KeyInfo keyInfo;

        try
        {
            keyInfo =
                LoadAndValidateKeyMetadata();
        }
        catch (Exception ex)
        {
            MessageBox.Show(
                $"Vendor Signing Key ไม่พร้อมใช้งาน\n\n{ex.Message}",
                "ManaChai License Generator",
                MessageBoxButtons.OK,
                MessageBoxIcon.Error);
            return;
        }

        string? password =
            PasswordDialog.Ask(
                this,
                "ใส่ Password ของ Vendor Private Key");

        if (password is null)
        {
            return;
        }

        try
        {
            DateTime issuedAtUtc =
                DateTime.UtcNow;

            bool isTrial =
                _trialRadio.Checked;

            var payload =
                new LicensePayload
                {
                    SchemaVersion =
                        "MCL-LIC-1",
                    KeyId =
                        keyInfo.KeyId,
                    LicenseId =
                        Guid.NewGuid()
                            .ToString("D")
                            .ToUpperInvariant(),
                    CustomerName =
                        customerName,
                    MachineId =
                        machineId,
                    LicenseType =
                        isTrial
                            ? "Trial"
                            : "Permanent",
                    IssuedAtUtc =
                        issuedAtUtc,
                    ExpiresAtUtc =
                        isTrial
                            ? issuedAtUtc.AddDays(7)
                            : null
                };

            byte[] canonicalBytes =
                Encoding.UTF8.GetBytes(
                    BuildCanonicalPayload(
                        payload));

            string privatePem =
                File.ReadAllText(
                    PrivateKeyPath);

            using RSA rsa =
                RSA.Create();

            rsa.ImportFromEncryptedPem(
                privatePem,
                password);

            byte[] signature =
                rsa.SignData(
                    canonicalBytes,
                    HashAlgorithmName.SHA256,
                    RSASignaturePadding.Pss);

            // ตรวจด้วย Public Key ก่อนเซฟเสมอ
            string publicPem =
                File.ReadAllText(
                    PublicKeyPath);

            using RSA publicRsa =
                RSA.Create();

            publicRsa.ImportFromPem(
                publicPem);

            bool signatureValid =
                publicRsa.VerifyData(
                    canonicalBytes,
                    signature,
                    HashAlgorithmName.SHA256,
                    RSASignaturePadding.Pss);

            if (!signatureValid)
            {
                throw new CryptographicException(
                    "ตรวจสอบ Signature หลังสร้างไม่ผ่าน");
            }

            var signedLicense =
                new SignedLicenseFile
                {
                    Payload =
                        payload,
                    SignatureAlgorithm =
                        "RSA-PSS-SHA256",
                    SignatureBase64 =
                        Convert.ToBase64String(
                            signature)
                };

            Directory.CreateDirectory(
                DefaultLicenseOutputDirectory);

            using var dialog =
                new SaveFileDialog
                {
                    Title =
                        "บันทึก ManaChai License",
                    Filter =
                        "ManaChai License (*.license)|*.license",
                    InitialDirectory =
                        DefaultLicenseOutputDirectory,
                    FileName =
                        BuildDefaultFileName(
                            machineId,
                            payload.LicenseType)
                };

            if (dialog.ShowDialog(this) !=
                DialogResult.OK)
            {
                return;
            }

            string destination =
                EnsureLicenseExtension(
                    dialog.FileName);

            string json =
                JsonSerializer.Serialize(
                    signedLicense,
                    new JsonSerializerOptions
                    {
                        WriteIndented = true
                    });

            File.WriteAllText(
                destination,
                json,
                new UTF8Encoding(
                    encoderShouldEmitUTF8Identifier: false));

            string expiryText =
                payload.ExpiresAtUtc.HasValue
                    ? payload.ExpiresAtUtc.Value
                        .ToLocalTime()
                        .ToString("dd/MM/yyyy HH:mm")
                    : "ไม่หมดอายุ";

            MessageBox.Show(
                "สร้าง License สำเร็จ\n\n" +
                $"ลูกค้า: {payload.CustomerName}\n" +
                $"Machine ID: {payload.MachineId}\n" +
                $"ประเภท: {payload.LicenseType}\n" +
                $"หมดอายุ: {expiryText}\n" +
                $"Key ID: {payload.KeyId}\n\n" +
                $"ไฟล์:\n{destination}",
                "ManaChai License Generator",
                MessageBoxButtons.OK,
                MessageBoxIcon.Information);
        }
        catch (CryptographicException)
        {
            MessageBox.Show(
                "Password ของ Vendor Private Key ไม่ถูกต้อง " +
                "หรือไฟล์ Private Key เสียหาย",
                "ManaChai License Generator",
                MessageBoxButtons.OK,
                MessageBoxIcon.Error);
        }
        catch (Exception ex)
        {
            MessageBox.Show(
                $"ไม่สามารถสร้าง License ได้\n\n{ex.Message}",
                "ManaChai License Generator",
                MessageBoxButtons.OK,
                MessageBoxIcon.Error);
        }
    }

    private static KeyInfo LoadAndValidateKeyMetadata()
    {
        if (!File.Exists(PrivateKeyPath) ||
            !File.Exists(PublicKeyPath) ||
            !File.Exists(KeyInfoPath))
        {
            throw new FileNotFoundException(
                "ไม่พบ Vendor Signing Key จาก Phase 2L.2");
        }

        KeyInfo? info =
            JsonSerializer.Deserialize<KeyInfo>(
                File.ReadAllText(
                    KeyInfoPath));

        if (info is null ||
            string.IsNullOrWhiteSpace(
                info.KeyId))
        {
            throw new InvalidDataException(
                "อ่าน key-info.json ไม่สำเร็จ");
        }

        string publicPem =
            File.ReadAllText(
                PublicKeyPath);

        using RSA publicRsa =
            RSA.Create();

        publicRsa.ImportFromPem(
            publicPem);

        byte[] publicBytes =
            publicRsa.ExportSubjectPublicKeyInfo();

        byte[] fingerprint =
            SHA256.HashData(
                publicBytes);

        string fingerprintText =
            Convert.ToHexString(
                fingerprint);

        string expectedKeyId =
            "MC-KEY-" +
            Convert.ToHexString(
                fingerprint.AsSpan(0, 6));

        if (!string.Equals(
                info.PublicKeySha256,
                fingerprintText,
                StringComparison.OrdinalIgnoreCase) ||
            !string.Equals(
                info.KeyId,
                expectedKeyId,
                StringComparison.Ordinal))
        {
            throw new CryptographicException(
                "Public Key ไม่ตรงกับ key-info.json");
        }

        return info;
    }

    private static string BuildCanonicalPayload(
        LicensePayload payload)
    {
        // รูปแบบนี้จะถูกใช้เหมือนกันใน Client Validator (Phase 2L.4)
        // ห้ามเปลี่ยนลำดับ field โดยไม่เปลี่ยน SchemaVersion
        return string.Join(
            "\n",
            new[]
            {
                $"SchemaVersion={payload.SchemaVersion}",
                $"KeyId={payload.KeyId}",
                $"LicenseId={payload.LicenseId}",
                $"CustomerName={EscapeCanonical(payload.CustomerName)}",
                $"MachineId={payload.MachineId}",
                $"LicenseType={payload.LicenseType}",
                $"IssuedAtUtc={payload.IssuedAtUtc.ToUniversalTime():O}",
                $"ExpiresAtUtc={(payload.ExpiresAtUtc.HasValue ? payload.ExpiresAtUtc.Value.ToUniversalTime().ToString("O") : string.Empty)}"
            });
    }

    private static string EscapeCanonical(
        string value)
    {
        return value
            .Replace("\\", "\\\\")
            .Replace("\r", "\\r")
            .Replace("\n", "\\n");
    }

    private static string NormalizeMachineId(
        string value)
    {
        return value
            .Trim()
            .ToUpperInvariant();
    }

    private static string BuildDefaultFileName(
        string machineId,
        string licenseType)
    {
        string machinePart =
            machineId.Replace(
                "-",
                string.Empty);

        return
            $"ManaChaiLeasing_{machinePart}_{licenseType}.license";
    }

    private static string EnsureLicenseExtension(
        string path)
    {
        return path.EndsWith(
                ".license",
                StringComparison.OrdinalIgnoreCase)
            ? path
            : path + ".license";
    }

        IWin32Window owner,
        string title)
    {
        using var dialog =
            new PasswordDialog(
                title);

        return dialog.ShowDialog(
                   owner) ==
               DialogResult.OK
            ? dialog._result
            : null;
    }
